cmake_minimum_required(VERSION 3.24)
project(LazarusShare VERSION 0.1.4 LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network WebSockets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.22 gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-sdp-1.0 gstreamer-webrtc-1.0)
add_executable(lazarus-share app/main.cpp app/window.cpp app/window.h app/protocol.cpp app/protocol.h app/peer.cpp app/peer.h app/capture.cpp app/capture.h app/audio.cpp app/audio.h)
target_link_libraries(lazarus-share PRIVATE Qt6::Widgets Qt6::Network Qt6::WebSockets PkgConfig::GST)
target_compile_definitions(lazarus-share PRIVATE GST_USE_UNSTABLE_API)
if(WIN32)
  target_link_libraries(lazarus-share PRIVATE ole32 uuid psapi)
  set_target_properties(lazarus-share PROPERTIES WIN32_EXECUTABLE TRUE)
else()
  find_package(Qt6 REQUIRED COMPONENTS DBus)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
  target_sources(lazarus-share PRIVATE app/portal.cpp app/portal.h)
  target_link_libraries(lazarus-share PRIVATE Qt6::DBus PkgConfig::PIPEWIRE)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(tls-test app/tls_test.cpp)
  target_link_libraries(tls-test PRIVATE Qt6::Core Qt6::Network Qt6::WebSockets)
  add_executable(protocol-test app/protocol_test.cpp app/protocol.cpp app/protocol.h)
  target_link_libraries(protocol-test PRIVATE Qt6::Core)
  add_test(NAME protocol COMMAND protocol-test)
  add_executable(media-test app/media_test.cpp app/peer.cpp app/peer.h app/protocol.cpp app/protocol.h)
  target_compile_definitions(media-test PRIVATE GST_USE_UNSTABLE_API)
  target_link_libraries(media-test PRIVATE Qt6::Core Qt6::Gui PkgConfig::GST)
  add_test(NAME media COMMAND media-test)
  add_test(NAME media-quality COMMAND media-test --stress-quality)
  add_test(NAME media-quality-cpu COMMAND media-test --stress-quality)
  set_tests_properties(media-quality-cpu PROPERTIES ENVIRONMENT "LAZARUS_VIDEO_ENCODER=vp8")
  add_test(NAME media-video-only COMMAND media-test --video-only)
  if(NOT WIN32)
    add_executable(audio-test app/audio_test.cpp app/audio.cpp app/audio.h)
    target_link_libraries(audio-test PRIVATE Qt6::Core PkgConfig::GST PkgConfig::PIPEWIRE)
    add_executable(portal-test app/portal_test.cpp app/portal.cpp app/portal.h)
    target_link_libraries(portal-test PRIVATE Qt6::Core Qt6::DBus)
    add_test(NAME portal COMMAND dbus-run-session -- $<TARGET_FILE:portal-test>)
    add_executable(session-test app/portal.cpp app/session_test.cpp app/window.cpp app/window.h app/peer.cpp app/peer.h app/capture.cpp app/capture.h app/audio.cpp app/audio.h app/protocol.cpp app/protocol.h)
    target_compile_definitions(session-test PRIVATE GST_USE_UNSTABLE_API)
    target_link_libraries(session-test PRIVATE Qt6::Widgets Qt6::Network Qt6::WebSockets Qt6::DBus PkgConfig::GST PkgConfig::PIPEWIRE)
  endif()
endif()
install(TARGETS lazarus-share RUNTIME DESTINATION bin)

foreach(target lazarus-share media-test session-test)
  if(TARGET ${target})
    target_sources(${target} PRIVATE app/encoder.cpp app/encoder.h)
  endif()
endforeach()
